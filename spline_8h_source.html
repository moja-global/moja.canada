219 void spline::set_coeffs_from_b()
220 {
221 assert(m_x.size() == m_y.size());
222 assert(m_x.size() == m_b.size());
223 assert(m_x.size() > 2);
224 size_t n = m_b.size();
225 if (m_c.size() != n)
226 m_c.resize(n);
227 if (m_d.size() != n)
228 m_d.resize(n);
229
230 for (size_t i = 0; i < n - 1; i++) {
231 const double h = m_x[i + 1] - m_x[i];
232 // from continuity and differentiability condition
233 m_c[i] = (3.0 * (m_y[i + 1] - m_y[i]) / h - (2.0 * m_b[i] + m_b[i + 1])) / h;
234 // from differentiability condition
235 m_d[i] = ((m_b[i + 1] - m_b[i]) / (3.0 * h) - 2.0 / 3.0 * m_c[i]) / h;
236 }
237
238 // for left extrapolation coefficients
239 m_c0 = (m_left == first_deriv) ? 0.0 : m_c[0];
240 }

242 void spline::set_points(const std::vector<double>& x,
243 const std::vector<double>& y,
244 spline_type type)
245 {
246 assert(x.size() == y.size());
247 assert(x.size() >= 3);
248 // not-a-knot with 3 points has many solutions
249 if (m_left == not_a_knot || m_right == not_a_knot)
250 assert(x.size() >= 4);
251 m_type = type;
252 m_made_monotonic = false;
253 m_x = x;
254 m_y = y;
255 int n = (int)x.size();
256 // check strict monotonicity of input vector x
257 for (int i = 0; i < n - 1; i++) {
258 assert(m_x[i] < m_x[i + 1]);
259 }
260
261
262 if (type == linear) {
263 // linear interpolation
264 m_d.resize(n);
265 m_c.resize(n);
266 m_b.resize(n);
267 for (int i = 0; i < n - 1; i++) {
268 m_d[i] = 0.0;
269 m_c[i] = 0.0;
270 m_b[i] = (m_y[i + 1] - m_y[i]) / (m_x[i + 1] - m_x[i]);
271 }
272 // ignore boundary conditions, set slope equal to the last segment
273 m_b[n - 1] = m_b[n - 2];
274 m_c[n - 1] = 0.0;
275 m_d[n - 1] = 0.0;
276 }
277 else if (type == cspline) {
278 // classical cubic splines which are C^2 (twice cont differentiable)
279 // this requires solving an equation system
280
281 // setting up the matrix and right hand side of the equation system
282 // for the parameters b[]
283 int n_upper = (m_left == spline::not_a_knot) ? 2 : 1;
284 int n_lower = (m_right == spline::not_a_knot) ? 2 : 1;
285 internal::band_matrix A(n, n_upper, n_lower);
286 std::vector<double> rhs(n);
287 for (int i = 1; i < n - 1; i++) {
288 A(i, i - 1) = 1.0 / 3.0 * (x[i] - x[i - 1]);
289 A(i, i) = 2.0 / 3.0 * (x[i + 1] - x[i - 1]);
290 A(i, i + 1) = 1.0 / 3.0 * (x[i + 1] - x[i]);
291 rhs[i] = (y[i + 1] - y[i]) / (x[i + 1] - x[i]) - (y[i] - y[i - 1]) / (x[i] - x[i - 1]);
292 }
293 // boundary conditions
294 if (m_left == spline::second_deriv) {
295 // 2*c[0] = f''
296 A(0, 0) = 2.0;
297 A(0, 1) = 0.0;
298 rhs[0] = m_left_value;
299 }
300 else if (m_left == spline::first_deriv) {
301 // b[0] = f', needs to be re-expressed in terms of c:
302 // (2c[0]+c[1])(x[1]-x[0]) = 3 ((y[1]-y[0])/(x[1]-x[0]) - f')
303 A(0, 0) = 2.0 * (x[1] - x[0]);
304 A(0, 1) = 1.0 * (x[1] - x[0]);
305 rhs[0] = 3.0 * ((y[1] - y[0]) / (x[1] - x[0]) - m_left_value);
306 }
307 else if (m_left == spline::not_a_knot) {
308 // f'''(x[1]) exists, i.e. d[0]=d[1], or re-expressed in c:
309 // -h1*c[0] + (h0+h1)*c[1] - h0*c[2] = 0
310 A(0, 0) = -(x[2] - x[1]);
311 A(0, 1) = x[2] - x[0];
312 A(0, 2) = -(x[1] - x[0]);
313 rhs[0] = 0.0;
314 }
315 else {
316 assert(false);
317 }
318 if (m_right == spline::second_deriv) {
319 // 2*c[n-1] = f''
320 A(n - 1, n - 1) = 2.0;
321 A(n - 1, n - 2) = 0.0;
322 rhs[n - 1] = m_right_value;
323 }
324 else if (m_right == spline::first_deriv) {
325 // b[n-1] = f', needs to be re-expressed in terms of c:
326 // (c[n-2]+2c[n-1])(x[n-1]-x[n-2])
327 // = 3 (f' - (y[n-1]-y[n-2])/(x[n-1]-x[n-2]))
328 A(n - 1, n - 1) = 2.0 * (x[n - 1] - x[n - 2]);
329 A(n - 1, n - 2) = 1.0 * (x[n - 1] - x[n - 2]);
330 rhs[n - 1] = 3.0 * (m_right_value - (y[n - 1] - y[n - 2]) / (x[n - 1] - x[n - 2]));
331 }
332 else if (m_right == spline::not_a_knot) {
333 // f'''(x[n-2]) exists, i.e. d[n-3]=d[n-2], or re-expressed in c:
334 // -h_{n-2}*c[n-3] + (h_{n-3}+h_{n-2})*c[n-2] - h_{n-3}*c[n-1] = 0
335 A(n - 1, n - 3) = -(x[n - 1] - x[n - 2]);
336 A(n - 1, n - 2) = x[n - 1] - x[n - 3];
337 A(n - 1, n - 1) = -(x[n - 2] - x[n - 3]);
338 rhs[0] = 0.0;
339 }
340 else {
341 assert(false);
342 }
343
344 // solve the equation system to obtain the parameters c[]
345 m_c = A.lu_solve(rhs);
346
347 // calculate parameters b[] and d[] based on c[]
348 m_d.resize(n);
349 m_b.resize(n);
350 for (int i = 0; i < n - 1; i++) {
351 m_d[i] = 1.0 / 3.0 * (m_c[i + 1] - m_c[i]) / (x[i + 1] - x[i]);
352 m_b[i] = (y[i + 1] - y[i]) / (x[i + 1] - x[i])
353 - 1.0 / 3.0 * (2.0 * m_c[i] + m_c[i + 1]) * (x[i + 1] - x[i]);
354 }
355 // for the right extrapolation coefficients (zero cubic term)
356 // f_{n-1}(x) = y_{n-1} + b*(x-x_{n-1}) + c*(x-x_{n-1})^2
357 double h = x[n - 1] - x[n - 2];
358 // m_c[n-1] is determined by the boundary condition
359 m_d[n - 1] = 0.0;
360 m_b[n - 1] = 3.0 * m_d[n - 2] * h * h + 2.0 * m_c[n - 2] * h + m_b[n - 2]; // = f'_{n-2}(x_{n-1})
361 if (m_right == first_deriv)
362 m_c[n - 1] = 0.0; // force linear extrapolation
363
364 }
365 else if (type == cspline_hermite) {
366 // hermite cubic splines which are C^1 (cont. differentiable)
367 // and derivatives are specified on each grid point
368 // (here we use 3-point finite differences)
369 m_b.resize(n);
370 m_c.resize(n);
371 m_d.resize(n);
372 // set b to match 1st order derivative finite difference
373 for (int i = 1; i < n - 1; i++) {
374 const double h = m_x[i + 1] - m_x[i];
375 const double hl = m_x[i] - m_x[i - 1];
376 m_b[i] = -h / (hl * (hl + h)) * m_y[i - 1] + (h - hl) / (hl * h) * m_y[i]
377 + hl / (h * (hl + h)) * m_y[i + 1];
378 }
379 // boundary conditions determine b[0] and b[n-1]
380 if (m_left == first_deriv) {
381 m_b[0] = m_left_value;
382 }
383 else if (m_left == second_deriv) {
384 const double h = m_x[1] - m_x[0];
385 m_b[0] = 0.5 * (-m_b[1] - 0.5 * m_left_value * h + 3.0 * (m_y[1] - m_y[0]) / h);
386 }
387 else if (m_left == not_a_knot) {
388 // f''' continuous at x[1]
389 const double h0 = m_x[1] - m_x[0];
390 const double h1 = m_x[2] - m_x[1];
391 m_b[0] = -m_b[1] + 2.0 * (m_y[1] - m_y[0]) / h0
392 + h0 * h0 / (h1 * h1) * (m_b[1] + m_b[2] - 2.0 * (m_y[2] - m_y[1]) / h1);
393 }
394 else {
395 assert(false);
396 }
397 if (m_right == first_deriv) {
398 m_b[n - 1] = m_right_value;
399 m_c[n - 1] = 0.0;
400 }
401 else if (m_right == second_deriv) {
402 const double h = m_x[n - 1] - m_x[n - 2];
403 m_b[n - 1] = 0.5 * (-m_b[n - 2] + 0.5 * m_right_value * h + 3.0 * (m_y[n - 1] - m_y[n - 2]) / h);
404 m_c[n - 1] = 0.5 * m_right_value;
405 }
406 else if (m_right == not_a_knot) {
407 // f''' continuous at x[n-2]
408 const double h0 = m_x[n - 2] - m_x[n - 3];
409 const double h1 = m_x[n - 1] - m_x[n - 2];
410 m_b[n - 1] = -m_b[n - 2] + 2.0 * (m_y[n - 1] - m_y[n - 2]) / h1 + h1 * h1 / (h0 * h0)
411 * (m_b[n - 3] + m_b[n - 2] - 2.0 * (m_y[n - 2] - m_y[n - 3]) / h0);
412 // f'' continuous at x[n-1]: c[n-1] = 3*d[n-2]*h[n-2] + c[n-1]
413 m_c[n - 1] = (m_b[n - 2] + 2.0 * m_b[n - 1]) / h1 - 3.0 * (m_y[n - 1] - m_y[n - 2]) / (h1 * h1);
414 }
415 else {
416 assert(false);
417 }
418 m_d[n - 1] = 0.0;
419
420 // parameters c and d are determined by continuity and differentiability
421 set_coeffs_from_b();
422
423 }
424 else {
425 assert(false);
426 }
427
428 // for left extrapolation coefficients
429 m_c0 = (m_left == first_deriv) ? 0.0 : m_c[0];
430 }

432 bool spline::make_monotonic()
433 {
434 assert(m_x.size() == m_y.size());
435 assert(m_x.size() == m_b.size());
436 assert(m_x.size() > 2);
437 bool modified = false;
438 const int n = (int)m_x.size();
439 // make sure: input data monotonic increasing --> b_i>=0
440 // input data monotonic decreasing --> b_i<=0
441 for (int i = 0; i < n; i++) {
442 int im1 = std::max(i - 1, 0);
443 int ip1 = std::min(i + 1, n - 1);
444 if (((m_y[im1] <= m_y[i]) && (m_y[i] <= m_y[ip1]) && m_b[i] < 0.0) ||
445 ((m_y[im1] >= m_y[i]) && (m_y[i] >= m_y[ip1]) && m_b[i] > 0.0)) {
446 modified = true;
447 m_b[i] = 0.0;
448 }
449 }
450 // if input data is monotonic (b[i], b[i+1], avg have all the same sign)
451 // ensure a sufficient criteria for monotonicity is satisfied:
452 // sqrt(b[i]^2+b[i+1]^2) <= 3 |avg|, with avg=(y[i+1]-y[i])/h,
453 for (int i = 0; i < n - 1; i++) {
454 double h = m_x[i + 1] - m_x[i];
455 double avg = (m_y[i + 1] - m_y[i]) / h;
456 if (avg == 0.0 && (m_b[i] != 0.0 || m_b[i + 1] != 0.0)) {
457 modified = true;
458 m_b[i] = 0.0;
459 m_b[i + 1] = 0.0;
460 }
461 else if ((m_b[i] >= 0.0 && m_b[i + 1] >= 0.0 && avg > 0.0) ||
462 (m_b[i] <= 0.0 && m_b[i + 1] <= 0.0 && avg < 0.0)) {
463 // input data is monotonic
464 double r = sqrt(m_b[i] * m_b[i] + m_b[i + 1] * m_b[i + 1]) / std::fabs(avg);
465 if (r > 3.0) {
466 // sufficient criteria for monotonicity: r<=3
467 // adjust b[i] and b[i+1]
468 modified = true;
469 m_b[i] *= (3.0 / r);
470 m_b[i + 1] *= (3.0 / r);
471 }
472 }
473 }
474
475 if (modified == true) {
476 set_coeffs_from_b();
477 m_made_monotonic = true;
478 }
479
480 return modified;
481 }

483 // return the closest idx so that m_x[idx] <= x (return 0 if x<m_x[0])
484 size_t spline::find_closest(double x) const
485 {
486 std::vector<double>::const_iterator it;
487 it = std::upper_bound(m_x.begin(), m_x.end(), x); // *it > x
488 size_t idx = std::max(int(it - m_x.begin()) - 1, 0); // m_x[idx] <= x
489 return idx;
490 }
491
492 double spline::operator() (double x) const
493 {
494 // polynomial evaluation using Horner's scheme
495 // TODO: consider more numerically accurate algorithms, e.g.:
496 // - Clenshaw
497 // - Even-Odd method by A.C.R. Newbery
498 // - Compensated Horner Scheme
499 size_t n = m_x.size();
500 size_t idx = find_closest(x);
501
502 double h = x - m_x[idx];
503 double interpol;
504 if (x < m_x[0]) {
505 // extrapolation to the left
506 interpol = (m_c0 * h + m_b[0]) * h + m_y[0];
507 }
508 else if (x > m_x[n - 1]) {
509 // extrapolation to the right
510 interpol = (m_c[n - 1] * h + m_b[n - 1]) * h + m_y[n - 1];
511 }
512 else {
513 // interpolation
514 interpol = ((m_d[idx] * h + m_c[idx]) * h + m_b[idx]) * h + m_y[idx];
515 }
516 return interpol;
517 }

519 double spline::deriv(int order, double x) const
520 {
521 assert(order > 0);
522 size_t n = m_x.size();
523 size_t idx = find_closest(x);
524
525 double h = x - m_x[idx];
526 double interpol;
527 if (x < m_x[0]) {
528 // extrapolation to the left
529 switch (order) {
530 case 1:
531 interpol = 2.0 * m_c0 * h + m_b[0];
532 break;
533 case 2:
534 interpol = 2.0 * m_c0;
535 break;
536 default:
537 interpol = 0.0;
538 break;
539 }
540 }
541 else if (x > m_x[n - 1]) {
542 // extrapolation to the right
543 switch (order) {
544 case 1:
545 interpol = 2.0 * m_c[n - 1] * h + m_b[n - 1];
546 break;
547 case 2:
548 interpol = 2.0 * m_c[n - 1];
549 break;
550 default:
551 interpol = 0.0;
552 break;
553 }
554 }
555 else {
556 // interpolation
557 switch (order) {
558 case 1:
559 interpol = (3.0 * m_d[idx] * h + 2.0 * m_c[idx]) * h + m_b[idx];
560 break;
561 case 2:
562 interpol = 6.0 * m_d[idx] * h + 2.0 * m_c[idx];
563 break;
564 case 3:
565 interpol = 6.0 * m_d[idx];
566 break;
567 default:
568 interpol = 0.0;
569 break;
570 }
571 }
572 return interpol;
573 }

575 std::vector<double> spline::solve(double y, bool ignore_extrapolation) const
576 {
577 std::vector<double> x; // roots for the entire spline
578 std::vector<double> root; // roots for each piecewise cubic
579 const size_t n = m_x.size();
580
581 // left extrapolation
582 if (ignore_extrapolation == false) {
583 root = internal::solve_cubic(m_y[0] - y, m_b[0], m_c0, 0.0, 1);
584 for (size_t j = 0; j < root.size(); j++) {
585 if (root[j] < 0.0) {
586 x.push_back(m_x[0] + root[j]);
587 }
588 }
589 }
590
591 // brute force check if piecewise cubic has roots in their resp. segment
592 // TODO: make more efficient
593 for (size_t i = 0; i < n - 1; i++) {
594 root = internal::solve_cubic(m_y[i] - y, m_b[i], m_c[i], m_d[i], 1);
595 for (size_t j = 0; j < root.size(); j++) {
596 double h = (i > 0) ? (m_x[i] - m_x[i - 1]) : 0.0;
597 double eps = internal::get_eps() * 512.0 * std::min(h, 1.0);
598 if ((-eps <= root[j]) && (root[j] < m_x[i + 1] - m_x[i])) {
599 double new_root = m_x[i] + root[j];
600 if (x.size() > 0 && x.back() + eps > new_root) {
601 x.back() = new_root; // avoid spurious duplicate roots
602 }
603 else {
604 x.push_back(new_root);
605 }
606 }
607 }
608 }
609
610 // right extrapolation
611 if (ignore_extrapolation == false) {
612 root = internal::solve_cubic(m_y[n - 1] - y, m_b[n - 1], m_c[n - 1], 0.0, 1);
613 for (size_t j = 0; j < root.size(); j++) {
614 if (0.0 <= root[j]) {
615 x.push_back(m_x[n - 1] + root[j]);
616 }
617 }
618 }
619
620 return x;
621 };

704 // LR-Decomposition of a band matrix
705 void band_matrix::lu_decompose()
706 {
707 int i_max, j_max;
708 int j_min;
709 double x;
710
711 // preconditioning
712 // normalize column i so that a_ii=1
713 for (int i = 0; i < this->dim(); i++) {
714 assert(this->operator()(i, i) != 0.0);
715 this->saved_diag(i) = 1.0 / this->operator()(i, i);
716 j_min = std::max(0, i - this->num_lower());
717 j_max = std::min(this->dim() - 1, i + this->num_upper());
718 for (int j = j_min; j <= j_max; j++) {
719 this->operator()(i, j) *= this->saved_diag(i);
720 }
721 this->operator()(i, i) = 1.0; // prevents rounding errors
722 }
723
724 // Gauss LR-Decomposition
725 for (int k = 0; k < this->dim(); k++) {
726 i_max = std::min(this->dim() - 1, k + this->num_lower()); // num_lower not a mistake!
727 for (int i = k + 1; i <= i_max; i++) {
728 assert(this->operator()(k, k) != 0.0);
729 x = -this->operator()(i, k) / this->operator()(k, k);
730 this->operator()(i, k) = -x; // assembly part of L
731 j_max = std::min(this->dim() - 1, k + this->num_upper());
732 for (int j = k + 1; j <= j_max; j++) {
733 // assembly part of R
734 this->operator()(i, j) = this->operator()(i, j) + x * this->operator()(k, j);
735 }
736 }
737 }
738 }
739 // solves Ly=b
740 std::vector<double> band_matrix::l_solve(const std::vector<double>& b) const
741 {
742 assert(this->dim() == (int)b.size());
743 std::vector<double> x(this->dim());
744 int j_start;
745 double sum;
746 for (int i = 0; i < this->dim(); i++) {
747 sum = 0;
748 j_start = std::max(0, i - this->num_lower());
749 for (int j = j_start; j < i; j++) sum += this->operator()(i, j) * x[j];
750 x[i] = (b[i] * this->saved_diag(i)) - sum;
751 }
752 return x;
753 }
754 // solves Rx=y
755 std::vector<double> band_matrix::r_solve(const std::vector<double>& b) const
756 {
757 assert(this->dim() == (int)b.size());
758 std::vector<double> x(this->dim());
759 int j_stop;
760 double sum;
761 for (int i = this->dim() - 1; i >= 0; i--) {
762 sum = 0;
763 j_stop = std::min(this->dim() - 1, i + this->num_upper());
764 for (int j = i + 1; j <= j_stop; j++) sum += this->operator()(i, j) * x[j];
765 x[i] = (b[i] - sum) / this->operator()(i, i);
766 }
767 return x;
768 }

770 std::vector<double> band_matrix::lu_solve(const std::vector<double>& b,
771 bool is_lu_decomposed)
772 {
773 assert(this->dim() == (int)b.size());

775 if (is_lu_decomposed == false) {
776 this->lu_decompose();
777 }
778 y = this->l_solve(b);
779 x = this->r_solve(y);
780 return x;

790 // solutions for a + b*x = 0
791 std::vector<double> solve_linear(double a, double b)
792 {
793 std::vector<double> x; // roots
794 if (b == 0.0) {
795 if (a == 0.0) {
796 // 0*x = 0
797 x.resize(1);
798 x[0] = 0.0; // any x solves it but we need to pick one
799 return x;
800 }
801 else {
802 // 0*x + ... = 0, no solution
803 return x;
804 }
805 }
806 else {
807 x.resize(1);
808 x[0] = -a / b;
809 return x;
810 }
811 }
812
813 // solutions for a + b*x + c*x^2 = 0
814 std::vector<double> solve_quadratic(double a, double b, double c,
815 int newton_iter = 0)
816 {
817 if (c == 0.0) {
818 return solve_linear(a, b);
819 }
820 // rescale so that we solve x^2 + 2p x + q = (x+p)^2 + q - p^2 = 0
821 double p = 0.5 * b / c;
822 double q = a / c;
823 double discr = p * p - q;
824 const double eps = 0.5 * internal::get_eps();
825 double discr_err = (6.0 * (p * p) + 3.0 * fabs(q) + fabs(discr)) * eps;
826
827 std::vector<double> x; // roots
828 if (fabs(discr) <= discr_err) {
829 // discriminant is zero --> one root
830 x.resize(1);
831 x[0] = -p;
832 }
833 else if (discr < 0) {
834 // no root
835 }
836 else {
837 // two roots
838 x.resize(2);
839 x[0] = -p - sqrt(discr);
840 x[1] = -p + sqrt(discr);
841 }
842
843 // improve solution via newton steps
844 for (size_t i = 0; i < x.size(); i++) {
845 for (int k = 0; k < newton_iter; k++) {
846 double f = (c * x[i] + b) * x[i] + a;
847 double f1 = 2.0 * c * x[i] + b;
848 // only adjust if slope is large enough
849 if (fabs(f1) > 1e-8) {
850 x[i] -= f / f1;
851 }
852 }
853 }
854
855 return x;
856 }
857
858 // solutions for the cubic equation: a + b*x +c*x^2 + d*x^3 = 0
859 // this is a naive implementation of the analytic solution without
860 // optimisation for speed or numerical accuracy
861 // newton_iter: number of newton iterations to improve analytical solution
862 // see also
863 // gsl: gsl_poly_solve_cubic() in solve_cubic.c
864 // octave: roots.m - via eigenvalues of the Frobenius companion matrix
865 std::vector<double> solve_cubic(double a, double b, double c, double d,
866 int newton_iter)
867 {
868 if (d == 0.0) {
869 return solve_quadratic(a, b, c, newton_iter);
870 }
871
872 // convert to normalised form: a + bx + cx^2 + x^3 = 0
873 if (d != 1.0) {
874 a /= d;
875 b /= d;
876 c /= d;
877 }
878
879 // convert to depressed cubic: z^3 - 3pz - 2q = 0
880 // via substitution: z = x + c/3
881 std::vector<double> z; // roots of the depressed cubic
882 double p = -(1.0 / 3.0) * b + (1.0 / 9.0) * (c * c);
883 double r = 2.0 * (c * c) - 9.0 * b;
884 double q = -0.5 * a - (1.0 / 54.0) * (c * r);
885 double discr = p * p * p - q * q; // discriminant
886 // calculating numerical round-off errors with assumptions:
887 // - each operation is precise but each intermediate result x
888 // when stored has max error of x*eps
889 // - only multiplication with a power of 2 introduces no new error
890 // - a,b,c,d and some fractions (e.g. 1/3) have rounding errors eps
891 // - p_err << |p|, q_err << |q|, ... (this is violated in rare cases)
892 // would be more elegant to use boost::numeric::interval<double>
893 const double eps = internal::get_eps();
894 double p_err = eps * ((3.0 / 3.0) * fabs(b) + (4.0 / 9.0) * (c * c) + fabs(p));
895 double r_err = eps * (6.0 * (c * c) + 18.0 * fabs(b) + fabs(r));
896 double q_err = 0.5 * fabs(a) * eps + (1.0 / 54.0) * fabs(c) * (r_err + fabs(r) * 3.0 * eps)
897 + fabs(q) * eps;
898 double discr_err = (p * p) * (3.0 * p_err + fabs(p) * 2.0 * eps)
899 + fabs(q) * (2.0 * q_err + fabs(q) * eps) + fabs(discr) * eps;
900
901 // depending on the discriminant we get different solutions
902 if (fabs(discr) <= discr_err) {
903 // discriminant zero: one or two real roots
904 if (fabs(p) <= p_err) {
905 // p and q are zero: single root
906 z.resize(1);
907 z[0] = 0.0; // triple root
908 }
909 else {
910 z.resize(2);
911 z[0] = 2.0 * q / p; // single root
912 z[1] = -0.5 * z[0]; // double root
913 }
914 }
915 else if (discr > 0) {
916 // three real roots: via trigonometric solution
917 z.resize(3);
918 double ac = (1.0 / 3.0) * acos(q / (p * sqrt(p)));
919 double sq = 2.0 * sqrt(p);
920 z[0] = sq * cos(ac);
921 z[1] = sq * cos(ac - 2.0 * M_PI / 3.0);
922 z[2] = sq * cos(ac - 4.0 * M_PI / 3.0);
923 }
924 else if (discr < 0.0) {
925 // single real root: via Cardano's fromula
926 z.resize(1);
927 double sgnq = (q >= 0 ? 1 : -1);
928 double basis = fabs(q) + sqrt(-discr);
929 double C = sgnq * pow(basis, 1.0 / 3.0); // c++11 has std::cbrt()
930 z[0] = C + p / C;
931 }
932 for (size_t i = 0; i < z.size(); i++) {
933 // convert depressed cubic roots to original cubic: x = z - c/3
934 z[i] -= (1.0 / 3.0) * c;
935 // improve solution via newton steps
936 for (int k = 0; k < newton_iter; k++) {
937 double f = ((z[i] + c) * z[i] + b) * z[i] + a;
938 double f1 = (3.0 * z[i] + 2.0 * c) * z[i] + b;
939 // only adjust if slope is large enough
940 if (fabs(f1) > 1e-8) {
941 z[i] -= f / f1;
942 }
943 }
944 }
945 // ensure if a=0 we get exactly x=0 as root
946 // TODO: remove this fudge
947 if (a == 0.0) {
948 assert(z.size() > 0); // cubic should always have at least one root
949 double xmin = fabs(z[0]);
950 size_t imin = 0;
951 for (size_t i = 1; i < z.size(); i++) {
952 if (xmin > fabs(z[i])) {
953 xmin = fabs(z[i]);
954 imin = i;
955 }
956 }
957 z[imin] = 0.0; // replace the smallest absolute value with 0
958 }
959 std::sort(z.begin(), z.end());
960 return z;
961 }
